# Bank — plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** App de ahorro "Bank" con fondos, movimientos (Meter · Sacar prestado · Devolver · Gastar), metas y préstamos con fecha — idéntica en pinta y animaciones a Facturas.

**Architecture:** Se copia `Facturas/index.html` (un solo archivo HTML+CSS+JS, ~10.3k líneas) a `Proyectos/Bank/` y se transforma: primero se agrega el modelo nuevo (fondos/movimientos) al lado del viejo, luego se reemplazan formulario y lista, y recién ahí se poda todo lo de facturas de un tirón (así la app carga en cada commit). **Tech Stack:** HTML + CSS + JS vanilla, Firebase Auth/Firestore v9 compat (CDN), jsPDF (carga perezosa), PWA (sw.js + manifest). Sin build.

**Spec:** `docs/superpowers/specs/2026-09-21-bank-design.md`

    const MOV_TYPES = { in: 'Meter', loan: 'Sacar prestado', repay: 'Devolver', out: 'Gastar' };
    const MOV_SIGN  = { in: 1, loan: -1, repay: 1, out: -1 };   // efecto de cada tipo sobre "Tienes"
    let funds = [];        // { id, name, goal, color, createdAt }
    let movements = [];    // { id, fundId, type, amount, date, note, dueDate, loanId }  — más reciente primero
    let PROFILE = { name: '' };
    let dashboardFundFilter = null;   // null = todos los fondos, o id de fondo

    // Solo AÑADE valores por defecto, nunca borra datos (backup viejo, doc corrupto en la nube).
    function normalizeFunds(arr) {
      if (!Array.isArray(arr)) return [];
      return arr.filter(f => f && typeof f === 'object' && f.id).map(f => {
        f.id = String(f.id);
        if (typeof f.name !== 'string') f.name = '';
        f.goal = Number(f.goal) || 0;
        if (typeof f.color !== 'string') f.color = '';
        if (typeof f.createdAt !== 'string') f.createdAt = '';
        return f;
      }).sort((a, b) => (a.createdAt || '').localeCompare(b.createdAt || '') || (a.id > b.id ? 1 : -1));
    }
    function normalizeMovements(arr) {
      if (!Array.isArray(arr)) return [];
      return arr.filter(m => m && typeof m === 'object' && m.id).map(m => {
        m.id = String(m.id);
        m.fundId = String(m.fundId || '');
        if (!MOV_TYPES[m.type]) m.type = 'in';
        m.amount = round2(m.amount);
        if (typeof m.date !== 'string') m.date = '';
        if (typeof m.note !== 'string') m.note = '';
        if (typeof m.dueDate !== 'string') m.dueDate = '';
        m.loanId = m.type === 'repay' ? String(m.loanId || '') : '';
        return m;
      });
    }
    // Orden canónico (fecha desc, luego id desc). Es el orden que se persiste y con el que se
    // comparan las huellas del listener en vivo — cualquier array de movimientos pasa por acá.
    function _sortMovements(arr = movements) {
      arr.sort((a, b) => (b.date || '').localeCompare(a.date || '') || (b.id > a.id ? 1 : -1));
      return arr;
    }

    // Saldo de un préstamo: monto − devoluciones que le apuntan. Se calcula siempre, no se guarda.
    // ponytail: O(n) por préstamo → fundBalance es O(n²); con miles de movimientos, indexar repays por loanId.
    function loanBalance(loan) {
      const repaid = round2(movements.filter(m => m.type === 'repay' && m.loanId === loan.id).reduce((s, m) => s + m.amount, 0));
      return { repaid, balance: round2(loan.amount - repaid) };
    }
    function isLoanSettled(loan) { return loanBalance(loan).balance <= 0; }
    function isLoanOverdue(m, today = todayISO()) {
      return m.type === 'loan' && !!m.dueDate && m.dueDate < today && !isLoanSettled(m);
    }
    // Tienes / Debes de un fondo (fundId null → todos los fondos juntos)
    function fundBalance(fundId) {
      let have = 0, owed = 0;
      for (const m of movements) {
        if (fundId && m.fundId !== fundId) continue;
        have += MOV_SIGN[m.type] * m.amount;
        if (m.type === 'loan') owed += loanBalance(m).balance;
      }
      return { have: round2(have), owed: round2(owed) };
    }
    function fundById(id) { return funds.find(f => f.id === id) || null; }
    function fundName(id) { return fundById(id)?.name || 'Sin fondo'; }
    // Préstamos con saldo pendiente de un fondo (para el selector de "Devolver")
    function openLoansOf(fundId) {
      return movements.filter(m => m.type === 'loan' && m.fundId === fundId && !isLoanSettled(m));
    }
    function _getHomeTab() { return localStorage.getItem('bank_home_tab') === 'history' ? 'history' : 'funds'; }

    // Movimientos dentro del período (y fondo) activos del Resumen — los usan métricas, gráfica y CSV
    function getDashboardMovements() {
      const range = dashboardDateRange();
      if (!range) return [];
      return movements.filter(m => {
        const d = new Date(m.date + 'T12:00:00');
        if (!(d >= range.startDate && d <= range.endDate)) return false;
        if (dashboardFundFilter && m.fundId !== dashboardFundFilter) return false;
        return true;
      });
    }

    // ── Carga / guardado local ──
    function saveFunds()     { return safeSetItem('bank_funds', JSON.stringify(funds)); }
    function saveMovements() { return safeSetItem('bank_movements', JSON.stringify(movements)); }
    function saveProfile()   { safeSetItem('bank_profile', JSON.stringify(PROFILE)); _syncSettings(); }

    // ── CRUD ──
    function addMovement(m) {
      m.id = Date.now().toString();
      movements.unshift(m); _sortMovements();
      if (!saveMovements()) { movements = movements.filter(x => x.id !== m.id); return false; }
      _syncMovement(m);
      return true;
    }
    function updateMovement(id, m) {
      const idx = movements.findIndex(x => x.id === id);
      if (idx === -1) return false;
      m.id = id;
      const before = movements.slice();   // copia de ANTES, para la copia automática
      movements[idx] = m; _sortMovements();
      if (!saveMovements()) { movements = before; return false; }
      takeSnapshot('antes-de-editar-movimiento', before);
      _syncMovement(m);
      return true;
    }
    // Devuelve lo que borró (el movimiento y, si era préstamo, sus devoluciones) para poder deshacer.
    function deleteMovementById(id) {
      const m = movements.find(x => x.id === id);
      if (!m) return [];
      const ids = [id].concat(m.type === 'loan' ? movements.filter(x => x.type === 'repay' && x.loanId === id).map(x => x.id) : []);
      const removed = movements.filter(x => ids.includes(x.id));
      movements = movements.filter(x => !ids.includes(x.id));
      saveMovements();
      ids.forEach(_deleteMovementCloud);
      return removed;
    }
    function addFund(f) {
      f.id = Date.now().toString(); f.createdAt = todayISO();
      funds.push(f);
      if (!saveFunds()) { funds.pop(); return false; }
      _syncFund(f);
      return true;
    }
    function updateFund(id, patch) {
      const f = fundById(id); if (!f) return false;
      Object.assign(f, patch); saveFunds(); _syncFund(f);
      return true;
    }
    function deleteFund(id) {
      funds = funds.filter(f => f.id !== id);
      const gone = movements.filter(m => m.fundId === id).map(m => m.id);
      movements = movements.filter(m => m.fundId !== id);
      saveFunds(); saveMovements();
      _deleteFundCloud(id); gone.forEach(_deleteMovementCloud);
    }

    const _fundRef = () => _db.collection('users').doc(_currentUser.uid).collection('funds');
    const _movRef  = () => _db.collection('users').doc(_currentUser.uid).collection('movements');
    const _setRef  = () => _db.collection('users').doc(_currentUser.uid).collection('settings').doc('main');

    // Con la persistencia offline habilitada, sin conexión estas promesas quedan pendientes
    // hasta reconectar — no caen en el catch. Si SÍ caen es un error real (permisos, doc
    // corrupto), por eso se avisa con toast.
    function _cloudSet(ref, obj, what) {
      if (!_currentUser) return;
      ref().doc(obj.id).set(JSON.parse(JSON.stringify(obj))).catch(e => {
        console.warn('sync ' + what + ':', e);
        showToast('No se pudo sincronizar ' + what + ' con la nube: ' + e.message, 'error', { duration: 5000 });
      });
    }
    function _cloudDelete(ref, id, what) {
      if (!_currentUser) return;
      ref().doc(id).delete().catch(e => {
        console.warn('del ' + what + ':', e);
        showToast('No se pudo borrar ' + what + ' en la nube: ' + e.message, 'error', { duration: 5000 });
      });

      if (!_currentUser) return;

      });

    // Ejecuta una lista de operaciones de batch en tandas de 450 (límite real de Firestore: 500).
    async function _commitOpsInChunks(ops) {
      for (let i = 0; i < ops.length; i += 450) {
        const batch = _db.batch();
        ops.slice(i, i + 450).forEach(op => op(batch));
        await batch.commit();
      }
    }
    // Borra TODO en Firestore y re-sube lo local — solo tras import/restore.
    async function _replaceFirestoreAll() {
      if (!_currentUser) return;
      try {
        const [fs, ms] = await Promise.all([_fundRef().get(), _movRef().get()]);
        const ops = fs.docs.map(d => b => b.delete(d.ref))
          .concat(ms.docs.map(d => b => b.delete(d.ref)))
          .concat(funds.map(f => b => b.set(_fundRef().doc(f.id), f)))
          .concat(movements.map(m => b => b.set(_movRef().doc(m.id), m)));

    // Huella de lo último procesado — el primer disparo de onSnapshot al conectar trae lo MISMO

    let _lastFundsFingerprint = null, _lastMovementsFingerprint = null;

    function _stopRealtimeSync() {
      [_unsubFunds, _unsubMovements, _unsubSettings].forEach(u => { if (u) u(); });
      _unsubFunds = _unsubMovements = _unsubSettings = null;
    }
    // Merge local↔nube de UNA colección: la nube gana en mismo id; lo que solo existe local se sube.
    async function _mergeCollection(ref, local, normalize) {
      const cloud = (await ref().get()).docs.map(d => d.data());
      if (cloud.length === 0) {
        if (local.length) await _commitOpsInChunks(local.map(x => b => b.set(ref().doc(x.id), x)));
        return local;
      }
      const merged = new Map();
      local.forEach(x => merged.set(x.id, x));
      cloud.forEach(x => merged.set(x.id, x));
      const cloudIds = new Set(cloud.map(x => x.id));
      const localOnly = local.filter(x => !cloudIds.has(x.id));
      if (localOnly.length) await _commitOpsInChunks(localOnly.map(x => b => b.set(ref().doc(x.id), x)));
      return normalize(Array.from(merged.values()));
    }
    async function _loadFromFirestore() {
      try {
        funds = await _mergeCollection(_fundRef, funds, normalizeFunds); saveFunds();
        movements = _sortMovements(await _mergeCollection(_movRef, movements, normalizeMovements)); saveMovements();
        _lastFundsFingerprint = JSON.stringify(funds);
        _lastMovementsFingerprint = JSON.stringify(movements);
        const sSnap = await _setRef().get();
        if (sSnap.exists) _applyCloudSettings(sSnap.data());
        else _syncSettings();
      } catch (e) { console.warn('_loadFromFirestore:', e); }
    }
    function _applyCloudSettings(s) {
      if (s.profile && typeof s.profile === 'object') { PROFILE = { name: '', ...s.profile }; safeSetItem('bank_profile', JSON.stringify(PROFILE)); }
      if (s.homeTab === 'funds' || s.homeTab === 'history') safeSetItem('bank_home_tab', s.homeTab);
    }
    // Engancha los listeners en vivo — separado de _startRealtimeSync() para reconectar tras
    // un backgrounding largo sin repetir la descarga completa.
    function _attachRealtimeListeners() {
      _unsubFunds = _fundRef().onSnapshot(snap => {
        if (snap.metadata.hasPendingWrites) return;   // eco de nuestra propia escritura
        const fresh = normalizeFunds(snap.docs.map(d => d.data()));
        const fp = JSON.stringify(fresh);
        if (fp === _lastFundsFingerprint) return;
        _lastFundsFingerprint = fp;
        funds = fresh; saveFunds(); _refreshCurrentView();
      }, e => console.warn('funds snapshot:', e));
      _unsubMovements = _movRef().onSnapshot(snap => {
        if (snap.metadata.hasPendingWrites) return;
        const fresh = _sortMovements(normalizeMovements(snap.docs.map(d => d.data())));
        const fp = JSON.stringify(fresh);
        if (fp === _lastMovementsFingerprint) return;
        _lastMovementsFingerprint = fp;
        movements = fresh; saveMovements(); _refreshCurrentView();
      }, e => console.warn('movements snapshot:', e));
      _unsubSettings = _setRef().onSnapshot(snap => {
        if (snap.exists) _applyCloudSettings(snap.data());
      }, e => console.warn('settings snapshot:', e));
    }
    async function _startRealtimeSync() {
      _stopRealtimeSync();
      await _loadFromFirestore();
      _attachRealtimeListeners();
    }
